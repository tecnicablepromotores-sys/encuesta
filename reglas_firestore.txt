rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }
    function isAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/usuarios/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.rol == 'administrador';
    }
    function isSupervisor() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/usuarios/$(request.auth.uid)) &&
        (get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.rol == 'supervisor' ||
         get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.rol == 'administrador');
    }
    function getUserZone() {
      return get(/databases/$(database)/documents/usuarios/$(request.auth.uid)).data.zone;
    }
    function esDueno(promotorUidCampo) {
      return isSignedIn() && request.auth.uid == promotorUidCampo;
    }

    // ===== USUARIOS (PROTEGIDOS: solo autenticados) =====
    // El formulario público NO lee aquí. Lee de promotores_publicos.
    match /usuarios/{uid} {
      allow read: if isSignedIn();
      allow create: if isSignedIn() && request.auth.uid == uid;
      allow update: if isAdmin() ||
        (isSupervisor() && request.resource.data.rol != 'administrador') ||
        (isSignedIn() && request.auth.uid == uid &&
         request.resource.data.rol == resource.data.rol);
      allow delete: if isAdmin();
    }

    // ===== PROMOTORES PÚBLICOS (solo nombre + rol, para el formulario) =====
    // Escrita por GitHub Actions (Admin SDK, ignora reglas).
    // Lectura pública para que el formulario sin login pueda leer.
    match /promotores_publicos/{promotorId} {
      allow read: if true;
      allow create, update, delete: if isSignedIn();
    }

    // ===== CENSO PROSPECTOS =====
    match /censo_prospectos/{docId} {
      allow read, update: if isAdmin() ||
        (isSupervisor() && resource.data.zone == getUserZone()) ||
        esDueno(resource.data.promotorUid);
      allow delete: if isAdmin() || esDueno(resource.data.promotorUid);
      allow create: if isSignedIn() && request.resource.data.promotorUid == request.auth.uid;
    }

    // ===== CLIENTES APP =====
    match /clientes_app/{docId} {
      allow read, update: if isAdmin() ||
        (isSupervisor() && resource.data.zone == getUserZone()) ||
        esDueno(resource.data.promotorUid);
      allow delete: if isAdmin() || esDueno(resource.data.promotorUid);
      allow create: if isSignedIn() && request.resource.data.promotorUid == request.auth.uid;
    }

    // ===== PORTAL WEB (formulario público sin login) =====
    match /portal_web/{docId} {
      allow create: if true;
      allow read, update: if isAdmin() || (isSupervisor() && resource.data.zone == getUserZone());
      allow delete: if isAdmin();
    }

    // ===== UBICACIONES EN TIEMPO REAL =====
    match /promotores_ubicacion/{uid} {
      allow read: if isSignedIn();
      allow write: if isSignedIn() && request.auth.uid == uid;
    }

    // Bloquea cualquier otra colección no contemplada
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
